-- 1. Display every column and event (4 rows).
SELECT *
FROM lab03.campus_events
ORDER BY event_id;
 event_id |         event_title         |      location      | capacity 
----------+-----------------------------+--------------------+----------
      401 | Research Poster Workshop    | Innovation Studio  |       28
      402 | Student Coding Circle       | Computer Lab B     |       24
      403 | Design Thinking Session     | Collaboration Room |       35
      404 | Digital Storytelling Meetup | Media Space        |       20
(4 rows)

-- 2. Choose two columns and sort event titles alphabetically.
SELECT event_title, location
FROM lab03.campus_events
ORDER BY event_title ASC;
         event_title         |      location      
-----------------------------+--------------------
 Design Thinking Session     | Collaboration Room
 Digital Storytelling Meetup | Media Space
 Research Poster Workshop    | Innovation Studio
 Student Coding Circle       | Computer Lab B
(4 rows)

-- 3. Retrieve two fields from each event.
SELECT event_id, capacity
FROM lab03.campus_events
ORDER BY event_id;
 event_id | capacity 
----------+----------
      401 |       28
      402 |       24
      403 |       35
      404 |       20
(4 rows)

-- 4. Find a specific event by its title (1 row).
SELECT event_title, location, capacity
FROM lab03.campus_events
WHERE event_title = 'Student Coding Circle';
      event_title      |    location    | capacity 
-----------------------+----------------+----------
 Student Coding Circle | Computer Lab B |       24
(1 row)

-- 5. Filter events with at least 25 available places (2 rows).
SELECT event_title, location, capacity
FROM lab03.campus_events
WHERE capacity >= 25
ORDER BY capacity DESC;
       event_title        |      location      | capacity 
--------------------------+--------------------+----------
 Design Thinking Session  | Collaboration Room |       35
 Research Poster Workshop | Innovation Studio  |       28
(2 rows)

-- 6. Sort event titles in reverse alphabetical order.
SELECT event_title, location
FROM lab03.campus_events
ORDER BY event_title DESC;
         event_title         |      location      
-----------------------------+--------------------
 Student Coding Circle       | Computer Lab B
 Research Poster Workshop    | Innovation Studio
 Digital Storytelling Meetup | Media Space
 Design Thinking Session     | Collaboration Room
(4 rows)

-- 7. Return the first two events in ID order (2 rows).
SELECT event_id, event_title
FROM lab03.campus_events
ORDER BY event_id
LIMIT 2;
 event_id |       event_title        
----------+--------------------------
      401 | Research Poster Workshop
      402 | Student Coding Circle
(2 rows)

-- 8. Show the three smallest events by capacity (3 rows).
SELECT event_title, capacity, location
FROM lab03.campus_events
ORDER BY capacity ASC, event_title ASC
LIMIT 3;
         event_title         | capacity |     location      
-----------------------------+----------+-------------------
 Digital Storytelling Meetup |       20 | Media Space
 Student Coding Circle       |       24 | Computer Lab B
 Research Poster Workshop    |       28 | Innovation Studio
(3 rows)
